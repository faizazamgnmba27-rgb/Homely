database/schema.sql SQLite / Drift 
-- 100% Offline-First Data Architecture
CREATE TABLE inventory_items (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    category TEXT NOT NULL,
    quantity REAL NOT NULL DEFAULT 1.0,
    unit TEXT NOT NULL DEFAULT 'pcs',
    expiry_date INTEGER,
    min_stock_level REAL NOT NULL DEFAULT 1.0,
    price REAL,
    created_at INTEGER NOT NULL
);

CREATE TABLE bills (
    id TEXT PRIMARY KEY,
    bill_name TEXT NOT NULL,
    category TEXT NOT NULL,
    provider TEXT NOT NULL,
    amount REAL NOT NULL,
    due_date INTEGER NOT NULL,
    frequency TEXT NOT NULL,
    status TEXT CHECK(status IN ('upcoming','paid','overdue'))
);

CREATE TABLE assets (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    category TEXT NOT NULL,
    brand TEXT,
    model TEXT,
    warranty_expiry INTEGER NOT NULL,
    next_service_date INTEGER
);
